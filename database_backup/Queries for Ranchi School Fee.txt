USE ranchi_school;

-- ====== Student Details by giving Student Id ======
 SELECT TOP 1 
                        s.student_code,
                        s.student_name,
                        s.student_id,
                        sc.class_code,
                        c.class_name,
                        sc.section_code,
                        sec.section_name,
                        sc.year_no,
                        sc.academic_year,
                        s.father_name,
                        s.mother_name,
                        s.sub_acct_code
                    FROM students s
                    INNER JOIN student_curr_data sc ON s.student_code = sc.student_code
                    INNER JOIN classes c ON sc.class_code = c.class_code
                    INNER JOIN sections sec ON sc.section_code = sec.section_code
                    WHERE s.student_id = '26/39'
                    AND sc.academic_year = (
                        SELECT MAX(academic_year) 
                        FROM student_curr_data 
                        WHERE student_code = s.student_code
                    );

-- ====== tuition_fee_due ======
SELECT 
                        CASE WHEN SUM(amt) > 0 THEN SUM(amt) ELSE -SUM(amt) END as tuition_fee_due,
                        CASE WHEN SUM(amt) > 0 THEN 'Dr' ELSE 'Cr' END as crdr
                    FROM (
                        SELECT 'Open Bal' as description, sub_acct_code, 
                               CASE WHEN cr_dr='Dr' THEN SUM(amount) ELSE -SUM(amount) END as amt
                        FROM journal_tran a, journal_doc jd 
                        WHERE a.school_code = '01' 
                          AND a.finance_year = jd.finance_year 
                          AND a.finance_year = 2026
                          AND a.doc_type_code = jd.doc_type_code 
                          AND a.doc_no = jd.doc_no 
                          AND jd.doc_date <= CONVERT(datetime, '2026-11-11', 120)
                          AND ACCT_CODE IN (SELECT acct_code FROM acct WHERE is_studentacct = 1) 
                          AND amount > 0
                          AND sub_acct_code = 'ST012086'
                        GROUP BY sub_acct_code, cr_dr
                        UNION ALL
                        SELECT 'Open Bal' as description, sub_acct_code, -SUM(ft.amount) as amt
                        FROM fee_receipt_doc fd, fee_receipt_tran ft 
                        WHERE ft.school_code = '01' 
                          AND ft.school_code = fd.school_code 
                          AND fd.finance_year = ft.finance_year
                          AND fd.finance_year = 2026
                          AND ft.doc_type_code = 'FR' 
                          AND ft.doc_type_code = fd.doc_type_code 
                          AND ft.doc_no = fd.doc_no 
                          AND fd.doc_date <= CONVERT(datetime, '2026-11-11', 120)
                          AND acct_code IN (SELECT acct_code FROM acct WHERE is_studentacct = 1)
                          AND sub_acct_code =  'ST012086'
                          AND approved_by <> '_Cancel' 
                          AND cr_no = 0
                        GROUP BY sub_acct_code
                    ) x;

-- ====== transportation_fee_due ======
SELECT 
                        CASE WHEN SUM(amt) > 0 THEN SUM(amt) ELSE -SUM(amt) END as transportation_fee_due,
                        CASE WHEN SUM(amt) > 0 THEN 'Dr' ELSE 'Cr' END as crdr
                    FROM (
                        SELECT 'Open Bal' as description, sub_acct_code,
                               CASE WHEN cr_dr='Dr' THEN SUM(amount) ELSE -SUM(amount) END as amt
                        FROM journal_tran a, journal_doc jd 
                        WHERE a.school_code = '01' 
                          AND a.finance_year = jd.finance_year 
                          AND a.finance_year = 2026
                          AND a.doc_type_code = jd.doc_type_code 
                          AND a.doc_no = jd.doc_no 
                          AND jd.doc_date <= CONVERT(datetime, '2026-11-11', 120)
                          AND ACCT_CODE IN (
                              SELECT DISTINCT dr_acct_code 
                              FROM fee_charge_acct fc, transport_routes tr 
                              WHERE fc.fee_type_code = tr.fee_code
                          )
                          AND amount > 0
                          AND sub_acct_code =  'ST012086'
                        GROUP BY sub_acct_code, cr_dr
                        UNION ALL
                        SELECT 'Open Bal' as description, sub_acct_code, -SUM(ft.amount) as amt
                        FROM fee_receipt_doc fd, fee_receipt_tran ft 
                        WHERE ft.school_code = '01' 
                          AND ft.school_code = fd.school_code 
                          AND fd.finance_year = ft.finance_year
                          AND fd.finance_year = 2026
                          AND ft.doc_type_code = 'FR' 
                          AND ft.doc_type_code = fd.doc_type_code 
                          AND ft.doc_no = fd.doc_no 
                          AND fd.doc_date <= CONVERT(datetime, '2026-11-11', 120)
                          AND acct_code IN (
                              SELECT DISTINCT dr_acct_code 
                              FROM fee_charge_acct fc, transport_routes tr 
                              WHERE fc.fee_type_code = tr.fee_code
                          )
                          AND sub_acct_code =  'ST012086'
                          AND approved_by <> '_Cancel' 
                          AND cr_no = 0
                        GROUP BY sub_acct_code
                    ) x;

-- ====== Tuition Fee Statement Qry ======
					SELECT 
                    student_code,
                    student_name,
                    student_id,
                    class_name,
                    section_code,
                    daysscholor_border,
                    studenttype_name,
                    COALESCE(refno, '-') as ref,
                    COALESCE(doc_date, admission_date) as ddate,
                    COALESCE(debits, 0) as charged,
                    COALESCE(credits, 0) as credit,
                    COALESCE(narration, '-') as description
                FROM (  
                    SELECT 
                        sub_acct_code,
                        admission_date,
                        sc.student_code,
                        student_name,
                        student_id,
                        class_name,
                        section_code,
                        daysscholor_border,
                        studenttype_name 
                    FROM students s, student_curr_data sc, classes c, student_types st 
                    WHERE sc.academic_year = 2026 
                        AND sub_acct_code = 'ST012086'
                        AND s.student_code = sc.student_code  
                        AND sc.class_code = c.class_code 
                        AND sc.studenttype_code = st.studenttype_code
                ) x
                LEFT JOIN (
                    SELECT 
                        sub_acct_code,
                        jv_no as refno,
                        d.doc_date,
                        CASE WHEN cr_dr='Dr' THEN amount ELSE 0 END as debits,
                        CASE WHEN cr_dr='Cr' THEN amount ELSE 0 END as credits,
                        t.narration 
                    FROM journal_tran t, journal_doc d  
                    WHERE t.finance_year = d.finance_year 
                        AND d.doc_type_code = t.doc_type_code 
                        AND d.doc_no = t.doc_no  
                        AND d.finance_year = 2026
                        AND acct_code IN (SELECT acct_code FROM acct WHERE is_studentacct = 1) 
                    UNION ALL  
                    SELECT 
                        sub_acct_code,
                        fd.doc_type_code + '-' + CONVERT(varchar, fd.doc_no) as refno,
                        fd.doc_date,
                        0 as debits,
                        ft.amount as credits,
                        ft.narration 
                    FROM fee_receipt_doc fd, fee_receipt_tran ft  
                    WHERE ft.school_code = '01' 
                        AND ft.school_code = fd.school_code   
                        AND fd.academic_year = ft.academic_year 
                        AND fd.finance_year = ft.finance_year 
                        AND fd.finance_year = 2026
                        AND ft.doc_type_code = 'FR'  
                        AND ft.doc_type_code = fd.doc_type_code 
                        AND ft.doc_no = fd.doc_no  
                        AND acct_code IN (SELECT acct_code FROM acct WHERE is_studentacct = 1) 
                        AND approved_by <> '_Cancel'  
                        AND cr_no = 0
                ) d ON x.sub_acct_code = d.sub_acct_code  
                ORDER BY x.student_code, doc_date, debits, credits;
